pub fn update(model, msg) {
  case msg {
    UserChangedTitle(title) -> #(
      Model(..model, title: title), // next model
      [save_title(title)],       // requested work
    )
    TitleSaved -> #(model, [])
    TitleSaveFailed ->
      #(model, [log("Save failed.")])
  }
}
html`
  <h1>${title(model)}</h1>
  <input
    .value=${title(model)}
    @input=${event => dispatch(
      new UserChangedTitle(event.currentTarget.value)
    )}
  >
`
export const saveTitle = title => ({
  perform: () => localStorage.setItem('title', title),
  onSuccess: () => titleSaved(),
  onFailure: () => titleSaveFailed(),
})

export const log = message => ({
  perform: () => console.log(message),
})
Model → view · effects → runner · messages → update

Return a pair: the next model and a list of effects. The runtime renders the model, then performs the work. [] means no work to run.

Views stay close to the web

Describe browser work with effects and ongoing listeners with subscriptions. Their messages return to the same update function.

An alternative, by design

Like MVU, but prefer writing templates?

Glimlit is for developers who enjoy the Model–View–Update approach of Elm and Gleam frameworks such as Lustre, and want another option for authoring views. It combines Gleam’s state model with lit-html rather than expressing every element as a Gleam function call.
